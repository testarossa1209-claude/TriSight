<script>

function log(s){document.getElementById("status").innerHTML+=s+"<br>"}

function clean(t){return t.replace(/```json|```/g,"").trim()}

function safeParse(t){
try{return JSON.parse(clean(t))}
catch(e){throw "JSON解析失敗"}
}

window.onload=function(){
let s=localStorage.getItem("trisight_office_keys")
if(s){
let k=JSON.parse(s)
geminiKey.value=k.gemini||""
claudeKey.value=k.claude||""
openaiKey.value=k.openai||""
saveKeys.checked=true
}
}

async function readFile(){
let f=fileInput.files[0]
if(!f) throw "ファイル未選択"

if(f.name.endsWith(".docx")){
let r=await mammoth.extractRawText({arrayBuffer:await f.arrayBuffer()})
return r.value
}

if(f.name.endsWith(".pdf")){
let pdf=await pdfjsLib.getDocument({data:await f.arrayBuffer()}).promise
let t=""
for(let i=1;i<=pdf.numPages;i++){
let p=await pdf.getPage(i)
let c=await p.getTextContent()
t+=c.items.map(x=>x.str).join("")
}
return t
}

throw "未対応形式"
}

async function geminiCall(text){

let prompt=`あなたは介護事業所の書類作成AIです。役割：既存運営規定に改正条項を機械的に反映する。判断・解釈は禁止。
JSONのみ出力。
{"revised_text":"","diff_list":[{"section":"","before":"","after":"","reason":""}]}
既存:${text}
改正:${lawInput.value}
根拠:${refInput.value}
事業所:${nameInput.value}（${typeInput.value}）/${cityInput.value}`

let r=await fetch(`https://generativelanguage.googleapis.com/v1/models/gemini-1.5-pro:generateContent?key=${geminiKey.value}`,{
method:"POST",
body:JSON.stringify({contents:[{parts:[{text:prompt}]}]})
})
let j=await r.json()

if(!j.candidates||!j.candidates[0]) throw "Gemini応答異常"

let g=safeParse(j.candidates[0].content.parts[0].text)

if(!g.diff_list||g.diff_list.length===0) throw "差分未生成"

return g
}

async function claudeCall(text){

let prompt=`あなたは検証AI。不備検出のみ。修正禁止。
JSONのみ。
{"issues":[{"type":"","location":"","description":""}],"warnings":[{"description":""}]}
文書:${text}`

let r=await fetch("https://api.anthropic.com/v1/messages",{
method:"POST",
headers:{
"x-api-key":claudeKey.value,
"anthropic-version":"2023-06-01",
"anthropic-dangerous-direct-browser-access":"true"
},
body:JSON.stringify({model:"claude-sonnet-4-20250514",max_tokens:1000,messages:[{role:"user",content:prompt}]})
})

let j=await r.json()

if(!j.content) throw "Claude応答異常"

return safeParse(j.content[0].text)
}

async function openaiCall(text){

let prompt=`行政判定AI。合否のみ。
JSONのみ。
{"verdict":"approved/rejected","risks":[{"level":"","description":""}],"notes":"","submission_advice":""}
文書:${text}`

let r=await fetch("https://api.openai.com/v1/chat/completions",{
method:"POST",
headers:{
"Authorization":"Bearer "+openaiKey.value,
"Content-Type":"application/json"
},
body:JSON.stringify({model:"gpt-4o",messages:[{role:"user",content:prompt}]})
})

let j=await r.json()

if(!j.choices||!j.choices[0]) throw "OpenAI応答異常"

let t=safeParse(j.choices[0].message.content)

if(!t.verdict) throw "判定不正"

return t
}

function createDoc(name,text){
const {Document,Paragraph,Packer}=docx
let doc=new Document({
sections:[{
children:[
new Paragraph(text),
new Paragraph("本書はAI支援により作成されており、最終確認は事業者責任において実施するものとする。自治体独自の条例が存在する場合は別途確認が必要である。")
]
}]
})
Packer.toBlob(doc).then(b=>saveAs(b,name))
}

function formatDate(){
let d=new Date()
return `${d.getFullYear()}年${d.getMonth()+1}月${d.getDate()}日`
}

function createTodoke(){
return `変更届出書

${formatDate()}

${cityInput.value}長 殿

事業所名：${nameInput.value}
所在地：${addrInput.value}
事業種別：${typeInput.value}

【変更内容】
運営規定改正

【変更理由】
法令改正

【添付】
修正版・変更インデックス`
}

async function run(){

try{

if(saveKeys.checked){
localStorage.setItem("trisight_office_keys",JSON.stringify({
gemini:geminiKey.value,
claude:claudeKey.value,
openai:openaiKey.value
}))
}

if(!nameInput.value||!cityInput.value||!lawInput.value||!refInput.value){
throw "入力不足"
}

log("開始")

let base=await readFile()

let g=await geminiCall(base)
log("生成OK")

let c=await claudeCall(g.revised_text)
if(c.issues.length>0) log("不備検出あり")

let t=await openaiCall(g.revised_text)

if(t.verdict==="rejected"){
log("提出不可")
return
}

createDoc("修正版.docx",g.revised_text)

let idx=g.diff_list.map(x=>`${x.section}\n${x.before}\n→${x.after}\n理由:${x.reason}`).join("\n\n")
createDoc("変更インデックス.docx",idx)

createDoc("変更届.docx",createTodoke())

log("完了")

}catch(e){
log("エラー:"+e)
}

}
